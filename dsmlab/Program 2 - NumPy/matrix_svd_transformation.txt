# Matrix transformation using Python and SVD
matrix = np.array(
    [
        [3, 1, 1],
        [-1, 3, 1],
    ],
    dtype=float,
)

u, singular_values, vt = np.linalg.svd(matrix, full_matrices=False)
singular_value_matrix = np.diag(singular_values)
reconstructed_matrix = u @ singular_value_matrix @ vt

print("\nOriginal matrix:\n", matrix)
print("U matrix:\n", u)
print("Singular values:", singular_values)
print("V-transpose matrix:\n", vt)
print("Reconstructed matrix:\n", reconstructed_matrix)
print(
    "Reconstruction is correct:",
    np.allclose(matrix, reconstructed_matrix),
)
