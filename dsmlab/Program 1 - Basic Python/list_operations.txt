# List operations: append, insert, delete, remove and copy
items = ["Python", "Java", "C++"]
print("Original list:", items)

items.append("JavaScript")
print("After append:", items)

items.insert(1, "C")
print("After insert:", items)

del items[2]
print("After delete:", items)

items.remove("JavaScript")
print("After remove:", items)

copied_items = items.copy()
print("Copied list:", copied_items)
