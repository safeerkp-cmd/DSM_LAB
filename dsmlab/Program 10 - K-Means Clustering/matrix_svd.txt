import numpy as np

A = np.array([[1, 2], [3, 4]])

U, S, VT = np.linalg.svd(A)

print("Original Matrix:")
print(A)

print("U Matrix:")
print(U)

print("Singular Values:")
print(S)

print("VT Matrix:")
print(VT)

S_matrix = np.diag(S)

A_new = U @ S_matrix @ VT

print("Reconstructed Matrix:")
print(A_new)
